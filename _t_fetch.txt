    async function fetchShopifyTraffic() {
      const since = document.getElementById('masterSince').value;
      const until = document.getElementById('masterUntil').value;
      const dateVal = until || since || document.getElementById('shopifyFetchDate').value;
      const msgEl = document.getElementById('shopifyFetchMsg');
      const tagEl = document.getElementById('shopifyFetchStatus');

      if (!dateVal) {
        _trafMsg(msgEl, tagEl, 'err', '⚠ Pick a date range in the navbar.');
        return;
      }

      _trafMsg(msgEl, tagEl, 'loading', '⏳ Fetching from Shopify…');

      try {
        // Use since/until if available, otherwise fall back to single date
        const qs = (since && until)
          ? '?since=' + encodeURIComponent(since) + '&until=' + encodeURIComponent(until)
          : '?date=' + encodeURIComponent(dateVal);
        const resp = await fetch('/api/traffic' + qs);

        if (!resp.ok) {
          const errText = await resp.text().catch(() => '');
          throw new Error('HTTP ' + resp.status + (errText ? ' — ' + errText.slice(0, 120) : ''));
        }

        const raw = await resp.json();

        // API returns {error: "..."} on failure
        if (raw && raw.error) throw new Error(raw.error);

        const rows = Array.isArray(raw) ? raw : [];
        if (rows.length === 0) throw new Error('No rows returned for ' + dateVal + '. Check the date has traffic data.');

        // ── Column name resolver (Shopify returns varying formats) ──
        function rv(r, ...names) {
          for (const n of names) { if (r[n] !== undefined && r[n] !== null) return r[n]; }
          // Also try case-insensitive partial match
          const keys = Object.keys(r);
          for (const n of names) {
            const nl = n.toLowerCase();
            const k = keys.find(k => k.toLowerCase() === nl || k.toLowerCase().replace(/[_ ]/g, '') === nl.replace(/[_ ]/g, ''));
            if (k !== undefined) return r[k];
          }
          return null;
        }

        // ── Reset TRAFFIC_META ──
        TRAFFIC_META = { pdp: { visitors: 0, sessions: 0, cartAdds: 0, checkout: 0 }, collection: { visitors: 0, sessions: 0, cartAdds: 0, checkout: 0 }, home: { visitors: 0, sessions: 0, cartAdds: 0, checkout: 0 }, other: { visitors: 0, sessions: 0, cartAdds: 0, checkout: 0 } };
        COLLECTION_DATA = [];
        HOME_DATA = [];

        // Log first row's keys for debugging
        if (rows.length > 0) console.log('[ShopifyFetch] Column keys:', Object.keys(rows[0]));

        // ── Classify all rows by page type ──
        // New query returns per-UTM-source rows, so multiple rows can share the same path.
        // We aggregate by path first for PDP/Collection/Home totals, but keep UTM detail.
        const productRows = [];
        const pdpAgg = {}; // slug → aggregated metrics
        let _shopifyTotals = null; // capture WITH TOTALS summary row
        rows.forEach(r => {
          const lpType = rv(r, 'landing_page_type', 'Landing page type') || '';
          const lpPath = rv(r, 'landing_page_path', 'Landing page path') || '';
          const pt = classifyPageType(lpType, lpPath);
          if (pt === 'summary') {
            // Capture the totals row from WITH TOTALS
            _shopifyTotals = {
              visitors: +(rv(r, 'online_store_visitors', 'Online store visitors')) || 0,
              sessions: +(rv(r, 'sessions', 'Sessions')) || 0,
              cartAdds: +(rv(r, 'sessions_with_cart_additions', 'Sessions with cart additions', 'Added to cart')) || 0,
              checkout: +(rv(r, 'sessions_that_reached_checkout', 'Sessions that reached checkout')) || 0,
              bounceRate: +(rv(r, 'bounce_rate', 'Bounce rate')) || 0,  // server already computed
              avgDuration: +(rv(r, 'average_session_duration', 'Average session duration')) || 0,
            };
            return;
          }
          const v = +(rv(r, 'online_store_visitors', 'Online store visitors')) || 0;
          const s = +(rv(r, 'sessions', 'Sessions')) || 0;
          const c = +(rv(r, 'sessions_with_cart_additions', 'Sessions with cart additions', 'Added to cart')) || 0;
          const ck = +(rv(r, 'sessions_that_reached_checkout', 'Sessions that reached checkout')) || 0;
          // bounceRate: computed server-side
          const bounceRate = +(rv(r, 'bounce_rate', 'Bounce rate')) || 0;
          const avgDuration = +(rv(r, 'average_session_duration', 'Average session duration')) || 0;
          const ppSession = +(rv(r, 'pageviews_per_session', 'Pageviews per session')) || 0;
          const utmSource = rv(r, 'utm_source', 'UTM source') || '';
          const utmMedium = rv(r, 'utm_medium', 'UTM medium') || '';
          const utmCampaign = rv(r, 'utm_campaign', 'UTM campaign') || '';
          const lpUrl = rv(r, 'landing_page_url', 'Landing page URL') || '';
          const atc = s > 0 ? Math.round(c / s * 1000) / 10 : 0;

          // Aggregate into TRAFFIC_META
          TRAFFIC_META[pt].visitors += v;
          TRAFFIC_META[pt].sessions += s;
          TRAFFIC_META[pt].cartAdds += c;
          TRAFFIC_META[pt].checkout += ck;

          // Store individual rows by type
          if (pt === 'pdp') {
            productRows.push(r);
            // Aggregate by slug for DATA merge
            const slug = String(lpPath || '').replace(/^\/products\//, '').replace(/\/$/, '').toLowerCase();
            if (!pdpAgg[slug]) pdpAgg[slug] = { visitors: 0, sessions: 0, cartAdds: 0, checkout: 0, bounceRate: 0, avgDuration: 0, ppSession: 0, _count: 0, utmSources: {} };
            const agg = pdpAgg[slug];
            agg.visitors += v; agg.sessions += s; agg.cartAdds += c; agg.checkout += ck;
            // Weighted averages for bounce/duration/ppSession
            agg.bounceRate += bounceRate * s; agg.avgDuration += avgDuration * s; agg.ppSession += ppSession * s;
            agg._count += s;
            // Track UTM sources
            if (utmSource) {
              if (!agg.utmSources[utmSource]) agg.utmSources[utmSource] = { sessions: 0, visitors: 0, cartAdds: 0 };
              agg.utmSources[utmSource].sessions += s;
              agg.utmSources[utmSource].visitors += v;
              agg.utmSources[utmSource].cartAdds += c;
            }
          } else if (pt === 'collection') {
            const p = String(lpPath || '');
            const name = p.replace(/^\/collections\//, '').replace(/\/$/, '').replace(/-/g, ' ').replace(/\b\w/g, x => x.toUpperCase()) || 'Collection';
            COLLECTION_DATA.push({ path: p, name: name, visitors: v, sessions: s, cartAdds: c, checkout: ck, atcPct: atc, leakage: c - ck, bounceRate: bounceRate, avgDuration: avgDuration, ppSession: ppSession, utmSource: utmSource, utmMedium: utmMedium, utmCampaign: utmCampaign, _page_type: 'collection' });
          } else if (pt === 'home') {
            HOME_DATA.push({ path: String(lpPath || '/'), name: 'Home Page', visitors: v, sessions: s, cartAdds: c, checkout: ck, atcPct: atc, leakage: c - ck, bounceRate: bounceRate, avgDuration: avgDuration, ppSession: ppSession, utmSource: utmSource, utmMedium: utmMedium, utmCampaign: utmCampaign, _page_type: 'home' });
          }
        });
        // Aggregate collection rows by path (multiple UTM rows per collection)
        const collAgg = {};
        COLLECTION_DATA.forEach(r => {
          const k = r.path;
          if (!collAgg[k]) collAgg[k] = { path: r.path, name: r.name, visitors: 0, sessions: 0, cartAdds: 0, checkout: 0, bounceRate: 0, avgDuration: 0, ppSession: 0, _count: 0, utmSources: {}, _page_type: 'collection' };
          const a = collAgg[k];
          a.visitors += r.visitors; a.sessions += r.sessions; a.cartAdds += r.cartAdds; a.checkout += r.checkout;
          a.bounceRate += r.bounceRate * r.sessions; a.avgDuration += r.avgDuration * r.sessions; a.ppSession += r.ppSession * r.sessions;
          a._count += r.sessions;
          if (r.utmSource) {
            if (!a.utmSources[r.utmSource]) a.utmSources[r.utmSource] = { sessions: 0, visitors: 0 };
            a.utmSources[r.utmSource].sessions += r.sessions;
            a.utmSources[r.utmSource].visitors += r.visitors;
          }
        });
        COLLECTION_DATA = Object.values(collAgg).map(a => {
          const sc = a._count || 1;
          const topUtm = Object.entries(a.utmSources).sort((x, y) => y[1].sessions - x[1].sessions).map(e => e[0]).slice(0, 3).join(', ') || 'direct';
          return { ...a, bounceRate: Math.round(a.bounceRate / sc * 10) / 10, avgDuration: Math.round(a.avgDuration / sc), ppSession: Math.round(a.ppSession / sc * 10) / 10, atcPct: a.sessions > 0 ? Math.round(a.cartAdds / a.sessions * 1000) / 10 : 0, leakage: a.cartAdds - a.checkout, _topUtm: topUtm };
        });
        // Aggregate home rows
        const homeAgg = { visitors: 0, sessions: 0, cartAdds: 0, checkout: 0, bounceRate: 0, avgDuration: 0, ppSession: 0, _count: 0, utmSources: {} };
        HOME_DATA.forEach(r => {
          homeAgg.visitors += r.visitors; homeAgg.sessions += r.sessions; homeAgg.cartAdds += r.cartAdds; homeAgg.checkout += r.checkout;
          homeAgg.bounceRate += r.bounceRate * r.sessions; homeAgg.avgDuration += r.avgDuration * r.sessions; homeAgg.ppSession += r.ppSession * r.sessions; homeAgg._count += r.sessions;
          if (r.utmSource) {
            if (!homeAgg.utmSources[r.utmSource]) homeAgg.utmSources[r.utmSource] = { sessions: 0, visitors: 0 };
            homeAgg.utmSources[r.utmSource].sessions += r.sessions;
            homeAgg.utmSources[r.utmSource].visitors += r.visitors;
          }
        });
        if (homeAgg.sessions > 0) {
          const hsc = homeAgg._count || 1;
          const homeTopUtm = Object.entries(homeAgg.utmSources).sort((x, y) => y[1].sessions - x[1].sessions).map(e => e[0]).slice(0, 3).join(', ') || 'direct';
          HOME_DATA = [{ path: '/', name: 'Home Page', visitors: homeAgg.visitors, sessions: homeAgg.sessions, cartAdds: homeAgg.cartAdds, checkout: homeAgg.checkout, atcPct: homeAgg.sessions > 0 ? Math.round(homeAgg.cartAdds / homeAgg.sessions * 1000) / 10 : 0, leakage: homeAgg.cartAdds - homeAgg.checkout, bounceRate: Math.round(homeAgg.bounceRate / hsc * 10) / 10, avgDuration: Math.round(homeAgg.avgDuration / hsc), ppSession: Math.round(homeAgg.ppSession / hsc * 10) / 10, _topUtm: homeTopUtm, utmSources: homeAgg.utmSources, _page_type: 'home' }];
        } else { HOME_DATA = []; }
        COLLECTION_DATA.sort((a, b) => b.sessions - a.sessions);

        // Store Shopify totals globally for accurate KPIs
        SHOPIFY_TOTALS = _shopifyTotals;

        if (productRows.length === 0 && Object.keys(pdpAgg).length === 0) throw new Error('No /products/ rows in response.');

        // ── Merge aggregated PDP data into DATA ──
        let matched = 0, zeroed = 0;
        DATA.forEach(p => {
          const agg = pdpAgg[p.slug.toLowerCase()];
          if (agg) {
            p.visitors = agg.visitors;
            p.sessions = agg.sessions;
            p.cartAdds = agg.cartAdds;
            p.checkout = agg.checkout;
            const sc = agg._count || 1;
            p.bounceRate = Math.round(agg.bounceRate / sc * 10) / 10;
            p.avgDuration = Math.round(agg.avgDuration / sc);
            p.ppSession = Math.round(agg.ppSession / sc * 10) / 10;
            p._utmSources = agg.utmSources;
            p._topUtm = Object.entries(agg.utmSources).sort((a, b) => b[1].sessions - a[1].sessions).map(e => e[0]).slice(0, 3).join(', ') || 'direct';
            p.atcPct = p.sessions > 0
              ? Math.round(p.cartAdds / p.sessions * 1000) / 10
              : 0;
            p.leakage = p.cartAdds - p.checkout;
            p.isAdLeak = p.cartAdds > 0 && p.checkout === 0
              && ['out_of_stock', 'low_stock', 'broken_stock'].includes(p.stockStatus);
            p.acqSignal = p.visitors >= 50 && p.cartAdds > 0 && p.stockStatus === 'in_stock';
            matched++;
          } else {
            p.visitors = 0; p.sessions = 0; p.cartAdds = 0;
            p.checkout = 0; p.atcPct = 0; p.leakage = 0;
            p.bounceRate = 0; p.avgDuration = 0; p.ppSession = 0;
            p._utmSources = {}; p._topUtm = '';
            p.isAdLeak = false; p.acqSignal = false;
            zeroed++;
          }
        });

        const note = zeroed > 0 ? ' · ' + zeroed + ' had 0 traffic' : '';
        const collNote = TRAFFIC_META.collection.sessions > 0 ? ' · ' + TRAFFIC_META.collection.sessions + ' coll sessions' : '';
        const homeNote = TRAFFIC_META.home.sessions > 0 ? ' · ' + TRAFFIC_META.home.sessions + ' home sessions' : '';
        _trafMsg(msgEl, tagEl, 'live',
          '✓ ' + dateVal + ' — ' + rows.length + ' total rows, ' + productRows.length + ' PDP, ' + matched + ' matched' + collNote + homeNote + note
        );
        tagEl.textContent = 'Live · ' + dateVal;

        // Re-render all pages with fresh data
        renderHome();
        renderTraf();
        renderAds();
        // (navMeta removed)

      } catch (err) {
        _trafMsg(msgEl, tagEl, 'err', '✗ ' + err.message);
        console.error('[ShopifyFetch]', err);
      }
    }

    function _trafMsg(msgEl, tagEl, state, text) {
      const colors = { live: 'var(--in)', err: 'var(--out)', loading: 'var(--traf)' };
      const labels = { live: 'Live', err: 'Error', loading: 'Fetching…' };
      tagEl.className = 'shopify-status ' + state;
      tagEl.textContent = labels[state] || state;
      msgEl.style.color = colors[state] || 'var(--mu2)';
      msgEl.textContent = text;
    }
